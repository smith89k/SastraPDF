# What's new window Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A "What's new in 1.1.0" window, as in AI Hub, that the main window shows once per version and ☰ › Help › What's New reopens.

**Architecture:** The notes and three pure rules live in a new classic script `app/notes.js` (`window.SastraNotes`). index.html builds the window next to About with the same `.modal-bg` / `.modal` code, remembers the last version seen in localStorage `sastra-notes-seen`, and marks `<html data-notes="shown|none">` once the start-up decision is made (tests wait on it). main.js adds the Help menu item. `scripts/check-notes.js` stops `npm run dist` and the release workflow when package.json's version has no notes.

**Tech Stack:** Electron 44.7.2, plain HTML/CSS/JS (no build step), Node `vm` for the build check, Playwright (`playwright-core`) for tests.

- [ ] **Step 4: Write `scripts/check-notes.js`**

```js
// Run first by `npm run dist` and the release workflow: stops the build when app/notes.js has no What's new
// entry for package.json's version, or an entry is broken — a release would otherwise ship an empty What's new
// and nobody would notice. `node scripts/check-notes.js [file]` checks another copy (tests).
const fs = require('fs'), path = require('path'), vm = require('vm');
const ROOT = path.join(__dirname, '..');
const file = path.resolve(process.argv[2] || path.join(ROOT, 'app', 'notes.js'));
const version = require(path.join(ROOT, 'package.json')).version;
const problems = [];
let N = null;
try { const window = {}; vm.runInNewContext(fs.readFileSync(file, 'utf8'), { window }, { filename: file }); N = window.SastraNotes || null; }
catch (e) { problems.push(`could not run it: ${e.message}`); }
if (N) {
  const list = Array.isArray(N.NOTES) ? N.NOTES : [], kinds = Array.isArray(N.KINDS) ? N.KINDS : [];
  if (!list.length) problems.push('there are no entries');
  else if (list[0].version !== version) problems.push(`the newest entry is ${list[0].version}, but package.json says ${version}: write its notes at the top`);
  const seen = new Set();
  for (const e of list) {
    const v = e && e.version;
    if (typeof v !== 'string' || !v) { problems.push('an entry has no version'); continue; }
    if (seen.has(v)) problems.push(`${v} is listed twice`);
    seen.add(v);
    if (!/^\d{4}-\d{2}-\d{2}$/.test(e.date || '')) problems.push(`${v} has no date (YYYY-MM-DD)`);
    if (!Array.isArray(e.notes) || !e.notes.length) { problems.push(`${v} has no notes`); continue; }
    for (const n of e.notes) {
      if (!kinds.includes(n && n.kind)) problems.push(`${v}: unknown kind "${n && n.kind}" (use ${kinds.join(', ')})`);
      if (typeof (n && n.text) !== 'string' || !n.text.trim()) problems.push(`${v}: a note has no text`);
    }
  }
} else if (!problems.length) problems.push('it does not set window.SastraNotes');
if (problems.length) {
  console.error(`What's new (${path.relative(ROOT, file)}) is not ready:\n` + problems.map(p => '  - ' + p).join('\n'));
  process.exit(1);
}
console.log(`What's new: ${N.NOTES.length} versions, newest ${N.NOTES[0].version} = package.json`);
```

const NOTES_SEEN = 'sastra-notes-seen';   // the last version whose notes were shown
const KIND_WORDS = { new:'New', improved:'Improved', fixed:'Fixed', removed:'Removed' };
const esc = s => String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
function notesList(notes){
  const K = SastraNotes.KINDS, sorted = notes.map((n, i) => [n, i]).sort((a, b) => K.indexOf(a[0].kind) - K.indexOf(b[0].kind) || a[1] - b[1]);
  return `<ul class="notes-list">${sorted.map(([n]) => `<li><span class="notes-kind notes-kind-${n.kind}">${KIND_WORDS[n.kind]}</span><span>${esc(n.text)}</span></li>`).join('')}</ul>`;
}
// false when another box is open (it never goes over one). A version with no entry (a hand-made build) shows every known one.
function openNotes(version){
  if ($('.modal-bg')) return false;
  const cur = SastraNotes.noteFor(version), past = SastraNotes.earlierNotes(version), back = document.activeElement;
  const bg = document.createElement('div'); bg.className='modal-bg';
  bg.innerHTML = `<div class="modal notes" role="dialog" aria-modal="true" aria-labelledby="notesTitle">
    <div class="notes-head"><h2 id="notesTitle">${cur ? `What's new in ${esc(version)}` : "What's new"}</h2>${cur ? `<div class="notes-date">${esc(cur.date)}</div>` : ''}</div>
    <div class="notes-body">${cur ? notesList(cur.notes) : ''}${cur && past.length ? '<button class="btn notes-more" data-more>See earlier versions ›</button>' : ''}${past.length ? `<div class="notes-earlier"${cur ? ' hidden' : ''}>${past.map(e => `<section class="notes-past"><h3>${esc(e.version)}<span>${esc(e.date)}</span></h3>${notesList(e.notes)}</section>`).join('')}</div>` : ''}</div>
    <div class="acts"><button class="btn primary" data-close>Close</button></div>
  </div>`;
  // closing it in any way counts as seen; a blocked storage only means it may show once more
  const done = () => { bg.remove(); document.removeEventListener('keydown', k, true); try { localStorage.setItem(NOTES_SEEN, version); } catch {} back?.focus?.(); };
  const k = e => { if (e.key==='Escape'){ e.stopPropagation(); done(); } };
  bg.addEventListener('click', e => {
    const more = e.target.closest('[data-more]');
    if (more) { more.hidden = true; const list = bg.querySelector('.notes-earlier'); list.hidden = false; list.scrollIntoView({ block:'start' }); bg.querySelector('[data-close]').focus({ preventScroll:true }); return; }
    if (e.target===bg || e.target.closest('[data-close]')) done();
  });
  document.addEventListener('keydown', k, true);
  document.body.appendChild(bg); bg.querySelector('[data-close]').focus({ preventScroll:true });

}

  let shown = false;
  try {
    const v = await desk.appVersion();

  } catch(e){ console.warn(e); }
  document.documentElement.dataset.notes = shown ? 'shown' : 'none';
}

  }
  return false;
};
